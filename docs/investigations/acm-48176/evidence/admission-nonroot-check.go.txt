const auditNS = "e2e-tests-discovery-audit"

// grants: stock system:authenticated restricted-v2/v3, plus optional nonroot-v2 for a subject.
type audit3Authz struct{ nonrootSubject, nonrootGroup string }

func (a audit3Authz) Authorize(_ context.Context, attrs authorizer.Attributes) (authorizer.Decision, string, error) {
	if attrs.GetVerb() != "use" || attrs.GetResource() != "securitycontextconstraints" {
		return authorizer.DecisionNoOpinion, "", nil
	}
	u := attrs.GetUser()
	inGroup := func(g string) bool {
		for _, x := range u.GetGroups() {
			if x == g {
				return true
			}
		}
		return false
	}
	switch attrs.GetName() {
	case "restricted-v2", "restricted-v3":
		if inGroup("system:authenticated") {
			return authorizer.DecisionAllow, "", nil
		}
	case "nonroot-v2":
		if (a.nonrootSubject != "" && u.GetName() == a.nonrootSubject) || (a.nonrootGroup != "" && inGroup(a.nonrootGroup)) {
			return authorizer.DecisionAllow, "", nil
		}
	}
	return authorizer.DecisionNoOpinion, "", nil
}

func TestACM48176Audit3(t *testing.T) {
	base := os.Getenv("ACM_AUDIT_DIR")
	var sccs []*securityv1.SecurityContextConstraints
	for _, p := range []string{"restricted-v2", "restricted-v3", "anyuid", "privileged", "nonroot-v2"} {
		b, err := os.ReadFile(filepath.Join(base, "scc", p+".yaml"))
		if err != nil {
			t.Fatal(err)
		}
		scc := &securityv1.SecurityContextConstraints{}
		if err := yaml.Unmarshal(b, scc); err != nil {
			t.Fatal(err)
		}
		sccs = append(sccs, scc)
	}
	type tc struct{ variant, grant string; authz audit3Authz }
	cases := []tc{
		{"original", "stock", audit3Authz{}},
		{"corrected", "stock", audit3Authz{}},
		{"original", "nonroot-v2 to default SA", audit3Authz{nonrootSubject: "system:serviceaccount:" + auditNS + ":default"}},
		{"original", "nonroot-v2 to group system:serviceaccounts", audit3Authz{nonrootGroup: "system:serviceaccounts"}},
		{"original", "nonroot-v2 to human user only", audit3Authz{nonrootSubject: "kube:admin"}},
	}
	for _, name := range []string{"nginx-demo", "netshoot"} {
		for _, c := range cases {
			b, err := os.ReadFile(filepath.Join(base, "manifests", c.variant, name+".json"))
			if err != nil {
				t.Fatal(err)
			}
			dep := &appsv1.Deployment{}
			if err := json.Unmarshal(b, dep); err != nil {
				t.Fatal(err)
			}
			ext := v1.Pod{ObjectMeta: metav1.ObjectMeta{Name: name + "-x", Namespace: auditNS, Labels: dep.Spec.Template.Labels}, Spec: dep.Spec.Template.Spec}
			ext.Spec.ServiceAccountName = "default"
			var pod core.Pod
			if err := conversion.Convert_v1_Pod_To_core_Pod(&ext, &pod, nil); err != nil {
				t.Fatal(err)
			}
			ns := createNamespaceForTest()
			ns.Name = auditNS
			ns.Annotations[securityv1.UIDRangeAnnotation] = "1001530000/10000"
			ns.Annotations[securityv1.SupplementalGroupsAnnotation] = "1001530000/10000"
			plugin := newTestAdmission(createSCCLister(t, sccs), createNamespaceLister(t, ns), c.authz)
			requester := &user.DefaultInfo{Name: "system:serviceaccount:kube-system:replicaset-controller", Groups: []string{"system:authenticated", "system:serviceaccounts", "system:serviceaccounts:kube-system"}}
			attrs := admission.NewAttributesRecord(&pod, nil, core.Kind("Pod").WithVersion("v1"), auditNS, pod.Name, core.Resource("pods").WithVersion("v1"), "", admission.Create, nil, false, requester)
			err = plugin.(admission.MutationInterface).Admit(context.Background(), attrs, nil)
			if err == nil {
				err = plugin.(admission.ValidationInterface).Validate(context.Background(), attrs, nil)
			}
			res := "ACCEPTED"
			detail := ""
			if err != nil {
				res = "REJECTED"
				detail = err.Error()
			} else {
				uid := pod.Spec.Containers[0].SecurityContext.RunAsUser
				if uid == nil {
					uid = pod.Spec.SecurityContext.RunAsUser
				}
				detail = fmt.Sprintf("scc=%s uid=%v", pod.Annotations[securityv1.ValidatedSCCAnnotation], *uid)
			}
			fmt.Printf("RESULT %-10s %-10s %-58s %s %s\n", name, c.variant, c.grant, res, detail)
		}
	}
}
