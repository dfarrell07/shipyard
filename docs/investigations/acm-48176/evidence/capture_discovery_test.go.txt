func TestCaptureContinuity(t *testing.T) {
	for _, mode := range []string{"resume", "stream-410", "http-410"} {
		t.Run(mode, func(t *testing.T) {
			ctx, cancel := context.WithTimeout(context.Background(), 8*time.Second)
			defer cancel()
			var mu sync.Mutex
			lists := 0
			var rvs, selectors []string
			server := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
				if r.URL.Path != "/api/v1/namespaces" {
					t.Errorf("unexpected path %s", r.URL.Path)
					w.WriteHeader(404)
					return
				}
				w.Header().Set("Content-Type", "application/json")
				selector := r.URL.Query().Get("labelSelector")
				if r.URL.Query().Get("watch") != "true" {
					mu.Lock()
					lists++
					selectors = append(selectors, selector)
					mu.Unlock()
					_ = json.NewEncoder(w).Encode(map[string]any{"apiVersion": "v1", "kind": "NamespaceList", "metadata": map[string]string{"resourceVersion": "10"}, "items": []any{}})
					return
				}
				mu.Lock()
				rvs = append(rvs, r.URL.Query().Get("resourceVersion"))
				selectors = append(selectors, selector)
				count := len(rvs)
				mu.Unlock()
				status := map[string]any{"apiVersion": "v1", "kind": "Status", "status": "Failure", "reason": "Expired", "code": 410, "message": "fixture RV expired"}
				if count == 2 && mode == "http-410" {
					w.WriteHeader(410)
					_ = json.NewEncoder(w).Encode(status)
					return
				}
				w.WriteHeader(200)
				w.(http.Flusher).Flush()
				if count == 2 && mode == "stream-410" {
					_ = json.NewEncoder(w).Encode(map[string]any{"type": "ERROR", "object": status})
					w.(http.Flusher).Flush()
					return
				}
				rv := "11"
				if count == 2 {
					rv = "12"
				}
				obj := &unstructured.Unstructured{Object: map[string]any{"apiVersion": "v1", "kind": "Namespace", "metadata": map[string]any{"name": "proof-" + rv, "resourceVersion": rv}}}
				_ = json.NewEncoder(w).Encode(map[string]any{"type": "ADDED", "object": obj})
				w.(http.Flusher).Flush()
				if count == 1 {
					return
				}
				<-r.Context().Done()
			}))
			defer server.Close()
			client, err := dynamic.NewForConfig(&rest.Config{Host: server.URL})
			if err != nil {
				t.Fatal(err)
			}
			reader, writer := io.Pipe()
			var metadata bytes.Buffer
			done := make(chan error, 1)
			go func() {
				err := capture(ctx, client.Resource(schema.GroupVersionResource{Version: "v1", Resource: "namespaces"}), "e2e-framework=discovery", writer, &metadata)
				_ = writer.Close()
				done <- err
			}()
			decoder := json.NewDecoder(reader)
			var received []string
			terminal := false
			for {
				var record struct {
					Phase  string         `json:"phase"`
					Type   string         `json:"type"`
					Object map[string]any `json:"object"`
				}
				err := decoder.Decode(&record)
				if err == io.EOF {
					break
				}
				if err != nil {
					t.Fatal(err)
				}
				if record.Phase != "watch_event" {
					continue
				}
				if record.Type == "ERROR" {
					terminal = true
					continue
				}
				received = append(received, record.Object["metadata"].(map[string]any)["resourceVersion"].(string))
				if mode == "resume" && len(received) == 2 {
					cancel()
				}
			}
			err = <-done
			if mode == "resume" {
				if err != nil || terminal || !reflect.DeepEqual(received, []string{"11", "12"}) {
					t.Fatalf("resume: received=%v terminal=%v err=%v", received, terminal, err)
				}
			} else {
				if err == nil || !strings.Contains(err.Error(), "terminal watch error") || !terminal || !reflect.DeepEqual(received, []string{"11"}) {
					t.Fatalf("410: received=%v terminal=%v err=%v", received, terminal, err)
				}
			}
			mu.Lock()
			if lists != 1 || !reflect.DeepEqual(rvs, []string{"10", "11"}) {
				t.Errorf("lists=%d watchRVs=%v", lists, rvs)
			}
			for _, selector := range selectors {
				if selector != "e2e-framework=discovery" {
					t.Errorf("selector lost: %q", selector)
				}
			}
			mu.Unlock()
			ready := strings.Count(metadata.String(), `"phase":"watch_established"`)
			expectedReady := 2
			if mode == "http-410" {
				expectedReady = 1
			}
			if ready != expectedReady {
				t.Errorf("accepted watch records=%d, expected %d", ready, expectedReady)
			}
		})
	}
}

type failedOutput struct{}

func (failedOutput) Write([]byte) (int, error) { return 0, io.ErrClosedPipe }

func TestCaptureRejectsIncompleteInitialListAndOutputFailure(t *testing.T) {
	for _, mode := range []string{"missing-rv", "zero-rv", "partial-list", "failed-output"} {
		t.Run(mode, func(t *testing.T) {
			watches := 0
			server := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
				w.Header().Set("Content-Type", "application/json")
				if r.URL.Query().Get("watch") == "true" {
					watches++
					w.WriteHeader(500)
					return
				}
				metadata := map[string]string{"resourceVersion": "10"}
				if mode == "missing-rv" {
					delete(metadata, "resourceVersion")
				}
				if mode == "zero-rv" {
					metadata["resourceVersion"] = "0"
				}
				if mode == "partial-list" {
					metadata["continue"] = "remaining"
				}
				_ = json.NewEncoder(w).Encode(map[string]any{"apiVersion": "v1", "kind": "PodList", "metadata": metadata, "items": []any{}})
			}))
			defer server.Close()
			client, err := dynamic.NewForConfig(&rest.Config{Host: server.URL})
			if err != nil {
				t.Fatal(err)
			}
			var output io.Writer = io.Discard
			if mode == "failed-output" {
				output = failedOutput{}
			}
			ctx, cancel := context.WithTimeout(context.Background(), time.Second)
			defer cancel()
			err = capture(ctx, client.Resource(schema.GroupVersionResource{Version: "v1", Resource: "pods"}), "", output, io.Discard)
			if err == nil || watches != 0 {
				t.Fatalf("err=%v watchRequests=%d", err, watches)
			}
		})
	}
}
