package main

import (
	"encoding/json"
	"fmt"
	"net/http"
	"net/http/httptest"
	"os"
	"strings"
	"sync"
	"time"

	"github.com/onsi/gomega"
	lh "github.com/submariner-io/lighthouse/test/e2e/framework"
	sy "github.com/submariner-io/shipyard/test/e2e/framework"
	appsv1 "k8s.io/api/apps/v1"
	corev1 "k8s.io/api/core/v1"
	metav1 "k8s.io/apimachinery/pkg/apis/meta/v1"
	"k8s.io/client-go/kubernetes"
	"k8s.io/client-go/rest"
	clientcmdapi "k8s.io/client-go/tools/clientcmd/api"
)

type observation struct {
	Paths   []string `json:"request_paths"`
	Case    string   `json:"case"`
	Builder string   `json:"builder"`
	Failure string   `json:"failure,omitempty"`
	Posts   int      `json:"deployment_posts"`
	UID     *int64   `json:"uid,omitempty"`
	Millis  int64    `json:"elapsed_ms"`
}

func run(name, body, builder string, slowAuth bool) (o observation) {
	o.Case = name
	o.Builder = builder
	var mu sync.Mutex
	prefix := ""
	if strings.HasPrefix(name, "prefixed-") {
		prefix = "/clusters/proxy"
	}
	started := time.Now()
	defer func() {
		if p := recover(); p != nil {
			o.Failure = fmt.Sprint(p)
		}
		o.Millis = time.Since(started).Milliseconds()
	}()
	server := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		mu.Lock()
		o.Paths = append(o.Paths, r.Method+" "+r.URL.Path)
		mu.Unlock()
		w.Header().Set("Content-Type", "application/json")
		if strings.TrimPrefix(r.URL.Path, prefix) == "/apis" {
			fmt.Fprint(w, body)
			return
		}
		if r.Method == "POST" {
			var dep appsv1.Deployment
			if err := json.NewDecoder(r.Body).Decode(&dep); err != nil {
				panic(err)
			}
			o.Posts++
			o.UID = dep.Spec.Template.Spec.SecurityContext.RunAsUser
			_ = json.NewEncoder(w).Encode(dep)
			return
		}
		_ = json.NewEncoder(w).Encode(corev1.PodList{TypeMeta: metav1.TypeMeta{Kind: "PodList", APIVersion: "v1"}, Items: []corev1.Pod{{Status: corev1.PodStatus{Phase: corev1.PodRunning}}}})
	}))
	defer server.Close()
	config := &rest.Config{Host: server.URL + prefix, ContentConfig: rest.ContentConfig{ContentType: "application/json", AcceptContentTypes: "application/json"}}
	if slowAuth {
		config.ExecProvider = &clientcmdapi.ExecConfig{Command: "python3", Args: []string{"-c", `import time; time.sleep(2); print('{"apiVersion":"client.authentication.k8s.io/v1","kind":"ExecCredential","status":{"token":"dummy-fixture-token"}}')`}, APIVersion: "client.authentication.k8s.io/v1", InteractiveMode: clientcmdapi.NeverExecInteractiveMode}
	}
	client, err := kubernetes.NewForConfig(config)
	if err != nil {
		panic(err)
	}
	sy.KubeClients = []*kubernetes.Clientset{client}
	sy.TestContext.OperationTimeout = 1
	f := &lh.Framework{Framework: sy.NewBareFramework("uid-proof")}
	f.Namespace = "uid-proof"
	if builder == "nginx" {
		f.NewNginxDeployment(sy.ClusterA)
	} else {
		f.NewNetShootDeploymentInNS(sy.ClusterA, "uid-proof")
	}
	return o
}
func main() {
	gomega.Default = gomega.NewGomega(func(message string, _ ...int) { panic(message) })
	var observations []observation
	for _, c := range []struct{ name, body string }{
		{"valid-openshift", `{"apiVersion":"v1","kind":"APIGroupList","groups":[{"name":"security.openshift.io"}]}`},
		{"valid-kubernetes", `{"apiVersion":"v1","kind":"APIGroupList","groups":[{"name":"apps"}]}`},
		{"prefixed-valid-openshift", `{"apiVersion":"v1","kind":"APIGroupList","groups":[{"name":"security.openshift.io"}]}`},
		{"empty-object", `{}`},
		{"json-null", `null`},
		{"missing-kind", `{"groups":[{"name":"security.openshift.io"}]}`},
		{"wrong-kind-status-success", `{"apiVersion":"v1","kind":"Status","status":"Success"}`},
		{"wrong-kind-pod-list", `{"apiVersion":"v1","kind":"PodList","items":[]}`},
		{"malformed-json", `{"apiVersion":`},
		{"empty-body", ``},
	} {
		for _, b := range []string{"nginx", "netshoot"} {
			observations = append(observations, run(c.name, c.body, b, false))
		}
	}
	observations = append(observations, run("slow-exec-auth", `{"apiVersion":"v1","kind":"APIGroupList","groups":[{"name":"apps"}]}`, "nginx", true))
	if err := json.NewEncoder(os.Stdout).Encode(observations); err != nil {
		panic(err)
	}
}
